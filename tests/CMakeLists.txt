function(nexus_add_unit_test target library)
    add_executable(${target}
        unit/${target}.cpp
    )
    target_link_libraries(${target} PRIVATE
        ${library}
        nexus_warnings
        nexus_sanitizers
    )
    add_test(NAME ${target} COMMAND ${target})
endfunction()

function(nexus_add_tls_integration_test name script timeout_seconds)
    add_test(
        NAME ${name}
        COMMAND ${Python3_EXECUTABLE}
                ${NEXUS_INTEGRATION_DIR}/with_tls.py
                ${NEXUS_INTEGRATION_DIR}/${script}
                $<TARGET_FILE:nexus-edge>
                $<TARGET_FILE:nexus-client>
    )
    set_tests_properties(${name} PROPERTIES TIMEOUT ${timeout_seconds})
endfunction()
